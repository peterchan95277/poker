    // --- 輔助函數：卡牌處理與驗證 ---


    /**

     * 將卡牌字串 (例如 'As') 轉換為物件 { rank: 14, suit: 's', str: 'AS' }

     * @param {string} cardStr - 卡牌字串

     * @returns {{rank: number, suit: string, str: string} | null} - 卡牌物件或 null (如果無效)

     */

    function parseCard(cardStr) {

        if (cardStr.length !== 2) return null;

        const rankStr = cardStr[0].toUpperCase();

        const suit = cardStr[1].toLowerCase();

        const rankIndex = RANKS.indexOf(rankStr);


        if (rankIndex === -1 || SUITS.indexOf(suit) === -1) return null;


        // 撲克牌等級 (A=14, K=13, ..., 2=2)

        const rank = rankIndex + 2; 


        return { rank, suit, str: cardStr.toUpperCase() };

    }


    /**

     * 建立一副完整的卡牌

     * @returns {Array<{rank: number, suit: string, str: string}>} - 完整的牌組

     */

    function createDeck() {

        const deck = [];

        for (const suit of SUITS) {

            for (let i = 0; i < RANKS.length; i++) {

                const rankStr = RANKS[i];

                const rank = i + 2;

                deck.push({ rank, suit, str: (rankStr + suit).toUpperCase() });

            }

        }

        return deck;

    }

    // --- 撲克牌手牌評估核心邏輯 ---

    const HAND_RANKS = {

        STRAIGHT_FLUSH: 9, FOUR_OF_A_KIND: 8, FULL_HOUSE: 7, FLUSH: 6, STRAIGHT: 5, 

        THREE_OF_A_KIND: 4, TWO_PAIR: 3, ONE_PAIR: 2, HIGH_CARD: 1

    };


    /**

     * 從 7 張牌中找出最好的 5 張牌組合，並返回其等級和踢牌 (Kickers)

     * @param {Array<{rank: number, suit: string, str: string}>} sevenCards - 7 張卡牌物件

     * @returns {{rank: number, kickers: number[]}} - 牌型結果

     */

    function getBestFiveCardHand(sevenCards) {

        // 1. 準備數據：排序後的牌階、牌階計數

        const sortedRanks = sevenCards.map(c => c.rank).sort((a, b) => b - a);

        const rankCounts = sortedRanks.reduce((acc, rank) => {

            acc[rank] = (acc[rank] || 0) + 1;

            return acc;

        }, {});

        

        // 按數量和牌階降序排列的計數列表

        const counts = Object.entries(rankCounts).map(([rank, count]) => ({ rank: parseInt(rank), count })).sort((a, b) => {

            if (b.count !== a.count) return b.count - a.count;

            return b.rank - a.rank;

        });


        // 2. 順子檢查

        const uniqueRanks = Array.from(new Set(sortedRanks));

        // 處理 A-5 順子 (A=14，但順子時 A=1)

        if (uniqueRanks.includes(14)) uniqueRanks.push(1);

        uniqueRanks.sort((a, b) => b - a);

        

        let straightHigh = 0;

        for (let i = 0; i <= uniqueRanks.length - 5; i++) {

            if (uniqueRanks[i] === uniqueRanks[i + 4] + 4) {

                straightHigh = uniqueRanks[i];

                break;

            }

        }

        const isStraight = straightHigh > 0;


        // 3. 同花檢查

        const suitCounts = sevenCards.reduce((acc, c) => {

            acc[c.suit] = acc[c.suit] || [];

            acc[c.suit].push(c.rank);

            return acc;

        }, {});

        let flushSuit = null;

        let flushRanks = [];

        for (const suit of SUITS) {

            if (suitCounts[suit] && suitCounts[suit].length >= 5) {

                flushSuit = suit;

                // 取出同花中的最高 5 張牌

                flushRanks = suitCounts[suit].sort((a, b) => b - a).slice(0, 5);

                break;

            }

        }

        const isFlush = flushSuit !== null;


        // 4. 判斷牌型 (從最強到最弱)

        

        // 4.1 皇家同花順 / 同花順

        if (isFlush) {

            const flushUniqueRanks = Array.from(new Set(flushRanks));

            if (flushUniqueRanks.includes(14)) flushUniqueRanks.push(1);

            flushUniqueRanks.sort((a, b) => b - a);

            

            let straightFlushHigh = 0;

            for (let i = 0; i <= flushUniqueRanks.length - 5; i++) {

                if (flushUniqueRanks[i] === flushUniqueRanks[i + 4] + 4) {

                    straightFlushHigh = flushUniqueRanks[i];

                    return { rank: HAND_RANKS.STRAIGHT_FLUSH, kickers: [straightFlushHigh] };

                }

            }

        }


        // 4.2 四條

        if (counts[0] && counts[0].count === 4) {

            const quadRank = counts[0].rank;

            const kicker = sortedRanks.find(r => r !== quadRank);

            return { rank: HAND_RANKS.FOUR_OF_A_KIND, kickers: [quadRank, kicker || 0] };

        }


        // 4.3 滿堂紅 (葫蘆)

        if (counts[0] && counts[0].count >= 3 && counts[1] && counts[1].count >= 2) {

            return { rank: HAND_RANKS.FULL_HOUSE, kickers: [counts[0].rank, counts[1].rank] };

        }


        // 4.4 同花

        if (isFlush) {

            return { rank: HAND_RANKS.FLUSH, kickers: flushRanks };

        }


        // 4.5 順子

        if (isStraight) {

            return { rank: HAND_RANKS.STRAIGHT, kickers: [straightHigh] };

        }


        // 4.6 三條

        if (counts[0] && counts[0].count === 3) {

            const threeRank = counts[0].rank;

            const kickers = sortedRanks.filter(r => r !== threeRank).slice(0, 2);

            return { rank: HAND_RANKS.THREE_OF_A_KIND, kickers: [threeRank, ...kickers] };

        }


        // 4.7 兩對

        if (counts[0] && counts[0].count === 2 && counts[1] && counts[1].count === 2) {

            const pair1Rank = counts[0].rank;

            const pair2Rank = counts[1].rank;

            const kicker = sortedRanks.find(r => r !== pair1Rank && r !== pair2Rank);

            return { rank: HAND_RANKS.TWO_PAIR, kickers: [pair1Rank, pair2Rank, kicker || 0] };

        }


        // 4.8 一對

        if (counts[0] && counts[0].count === 2) {

            const pairRank = counts[0].rank;

            const kickers = sortedRanks.filter(r => r !== pairRank).slice(0, 3);

            return { rank: HAND_RANKS.ONE_PAIR, kickers: [pairRank, ...kickers] };

        }


        // 4.9 高牌

        return { rank: HAND_RANKS.HIGH_CARD, kickers: sortedRanks.slice(0, 5) };

    }


    /**

     * 比較兩手牌的評估結果

     * @returns {number} - 1 (Hand 1 勝), -1 (Hand 2 勝), 0 (和局)

     */

    function compareHands(hand1, hand2) {

        if (hand1.rank !== hand2.rank) {

            return hand1.rank > hand2.rank ? 1 : -1;

        }

        for (let i = 0; i < hand1.kickers.length; i++) {

            if (hand1.kickers[i] !== hand2.kickers[i]) {

                return hand1.kickers[i] > hand2.kickers[i] ? 1 : -1;

            }

        }

        return 0;

    }


    // --- Monte Carlo 模擬主函數 ---


    /**

     * 執行 Monte Carlo 模擬並計算勝率

     */

    async function calculateEquity() {

        // 1. 重設 UI 和錯誤訊息

        errorMsgEl.classList.add('hidden');

        resultsDisplay.classList.add('hidden');

        loadingSpinner.classList.remove('hidden');


        const numPlayers = parseInt(numPlayersEl.value);

        const p1Cards = player1CardStrs.filter(s => s !== '').map(parseCard);

        const rawBoardCards = boardCardStrs;

        const p2CardsProvided = player2CardStrs.filter(s => s !== '').length === 2;


        try {

            // 2. 驗證輸入數量

            if (p1Cards.length !== 2) {

                throw new Error('玩家 1 必須剛好選取 2 張手牌。');

            }

            if (player2CardStrs.filter(s => s !== '').length === 1) {

                throw new Error('玩家 2 的手牌必須是 2 張或 0 張。');

            }


            // 檢查板牌是否連續

            let boardCards = [];

            let hasEmptySlot = false;

            for(const cardStr of rawBoardCards) {

                if (cardStr === '') {

                    hasEmptySlot = true;

                } else if (hasEmptySlot) {

                    throw new Error('公共牌不能有中間的空牌插槽。請確保按順序選牌。');

                } else {

                    boardCards.push(parseCard(cardStr));

                }

            }

            

            // 3. 確定對手數量和類型

            let knownOpponentHands = [];

            if (p2CardsProvided) {

                knownOpponentHands.push(player2CardStrs.filter(s => s !== '').map(parseCard));

            }


            // 總對手數 = 總玩家數 - 1 (P1)

            const totalOpponents = numPlayers - 1; 

            // 未知對手數 = 總對手數 - 已知對手數 (P2)

            const unknownOpponentCount = totalOpponents - knownOpponentHands.length; 

            

            if (unknownOpponentCount < 0) {

                throw new Error('總玩家人數少於已輸入的玩家手牌數量。');

            }

            

            // 4. 檢查重複卡牌

            const knownCardStrs = getAllSelectedCardStrs();

            const uniqueCardStrs = new Set(knownCardStrs);

            if (uniqueCardStrs.size !== knownCardStrs.length) {

                throw new Error('輸入卡牌中存在重複的牌。請檢查您的選取。');

            }

            

            // 5. 建立未知牌組 (Deck)

            const deck = createDeck();

            const usedCardStrs = new Set(knownCardStrs);

            const unknownDeck = deck.filter(card => !usedCardStrs.has(card.str.toUpperCase()));

            

            // 確保牌組足夠

            const requiredCards = (2 * unknownOpponentCount) + (5 - boardCards.length);

            if (unknownDeck.length < requiredCards) {

                throw new Error(`牌組中剩餘的牌 (${unknownDeck.length}) 不足以模擬所有未知手牌和公共牌 (${requiredCards})。請檢查輸入是否過多或玩家數設置錯誤。`);

            }


            // 6. 執行模擬

            let wins = 0;

            let ties = 0;

            let losses = 0;


            const cardsNeededOnBoard = 5 - boardCards.length;

            

            await new Promise(resolve => setTimeout(resolve, 10)); // 讓 UI 有時間更新


            for (let i = 0; i < NUM_ITERATIONS; i++) {

                let availableCards = [...unknownDeck];

                

                // 簡單的洗牌 (Fisher-Yates)

                for (let j = availableCards.length - 1; j > 0; j--) {

                    const k = Math.floor(Math.random() * (j + 1));

                    [availableCards[j], availableCards[k]] = [availableCards[k], availableCards[j]];

                }

                

                // 發出剩餘的公共牌

                const dealtBoardCards = availableCards.splice(0, cardsNeededOnBoard);

                const finalBoard = [...boardCards, ...dealtBoardCards];


                // 發出所有未知對手的手牌

                let dealtOpponentHands = [];

                for (let j = 0; j < unknownOpponentCount; j++) {

                    // 每手牌 2 張

                    dealtOpponentHands.push(availableCards.splice(0, 2));

                }


                // 合併所有對手的手牌 (已知 + 未知)

                const allOpponentHands = [...knownOpponentHands, ...dealtOpponentHands];


                // 比較 P1 對抗所有對手

                const p1SevenCards = [...p1Cards, ...finalBoard];

                const p1HandRank = getBestFiveCardHand(p1SevenCards);

                

                let bestOpponentRank = null;

                

                // 找出所有對手中最強的一手牌 (只有最強對手才能阻止 P1 獲勝)

                for (const opponentHand of allOpponentHands) {

                    const opponentSevenCards = [...opponentHand, ...finalBoard];

                    const opponentHandRank = getBestFiveCardHand(opponentSevenCards);


                    if (bestOpponentRank === null) {

                        bestOpponentRank = opponentHandRank;

                    } else {

                        const comparison = compareHands(opponentHandRank, bestOpponentRank);

                        if (comparison === 1) {

                            // 發現更強的對手

                            bestOpponentRank = opponentHandRank;

                        }

                        // 如果 comparison === 0 (平手)，則維持現有的 bestOpponentRank

                    }

                }


                if (bestOpponentRank === null) {

                    // 這只會在 totalOpponents = 0 時發生，但我們驗證了 totalOpponents >= 1

                    wins++; 

                } else {

                    const finalResult = compareHands(p1HandRank, bestOpponentRank);

                    

                    if (finalResult === 1) {

                        wins++; // P1 嚴格擊敗最強對手

                    } else if (finalResult === 0) {

                        // P1 與最強對手平手。在多對手情況下，P1 分享彩池

                        ties++; 

                    } else {

                        losses++; // P1 輸給最強對手

                    }

                }

            }

            

            // 7. 顯示結果

            const total = wins + ties + losses;

            const winRate = (wins / total * 100).toFixed(2);

            const tieRate = (ties / total * 100).toFixed(2);

            const lossRate = (losses / total * 100).toFixed(2);


            winRateEl.textContent = `${winRate}%`;

            tieRateEl.textContent = `${tieRate}%`;

            lossRateEl.textContent = `${lossRate}%`;


            loadingSpinner.classList.add('hidden');

            resultsDisplay.classList.remove('hidden');


        } catch (error) {

            loadingSpinner.classList.add('hidden');

            errorMsgEl.textContent = `錯誤：${error.message}`;

            errorMsgEl.classList.remove('hidden');

        }

    }


    // --- 事件監聽器和初始化 ---


    document.addEventListener('DOMContentLoaded', () => {

         resultsDisplay.classList.add('hidden');

         calculateBtn.addEventListener('click', calculateEquity);

         

         // 初始化時渲染所有下拉選單

         renderAllDropdowns();

         

         // 確保在人數變動時，牌局顯示正確 (儘管邏輯上只有渲染會改變)

         numPlayersEl.addEventListener('change', renderAllDropdowns);

    });


</script>
